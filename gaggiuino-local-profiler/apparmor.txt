#include <tunables/global>

# GLP — Gaggiuino Local Profiler AppArmor profile.
#
# MUST BE VALIDATED IN complain MODE BEFORE THIS SHIPS AT enforce. See the
# PR description for the exact steps. A too-tight profile fails in the
# field with opaque errors (files silently unreadable, sockets silently
# refused) and this repo has no way to run real AppArmor mediation in CI —
# only a Home Assistant OS install can confirm it.
#
# VALIDATION STILL REQUIRED ON REAL HARDWARE: nothing in this repo/CI can
# exercise real AppArmor mediation, so this file's rules were written by
# inspection of the runtime shape below, never run in complain mode. Before
# any release relies on this profile at enforce, it MUST be tested in
# complain mode on real Home Assistant OS hardware and any denials it logs
# addressed — do NOT flip it to enforce on the strength of a PR alone.
#
# Runtime shape this profile is derived from (not a template) — the Go
# backend on Alpine, the only runtime as of the #977 cutover (the last
# Node source was removed from the tree in 3.0.0, #1028):
#   - Base image alpine:3.22 running the Go backend (go/) — see the
#     repo-root Dockerfile. No s6-overlay, init: false (config.yaml).
#     docker-entrypoint.sh starts as root, `chown -R glp:glp /data` (the
#     bind mount's owning UID varies per install, see #271), then
#     `exec su-exec glp "$@"` to run `glp-server` unprivileged for the rest
#     of the process lifetime. No node_modules/native-addon tree exists
#     (modernc.org/sqlite and the Go stdlib are pure Go, no CGo), and the
#     shell/exec chain is Alpine's busybox (/bin, /sbin) rather than
#     Debian's coreutils/dash — the broad `file,`/`ix` rules below cover
#     both but were not narrowed to busybox specifically.
#   - All persistent state lives under /data: the SQLite DB (glp.db, plus
#     its -wal/-shm files — go/internal/db opens with `PRAGMA journal_mode
#     = WAL`), a one-time glp.db.pre-go-backup
#     snapshot (see docker-entrypoint.sh), legacy per-entity JSON files
#     kept only as one-time migration sources, api_token.txt,
#     preheat_state.json, options.json, profiles_cache.json, and
#     bean-images/ (bean photos, downloaded from an allow-listed set of
#     import hosts or uploaded by the user). Backups are streamed out over
#     HTTP — nothing is written to a temp/backup directory.
#   - The Go binary is statically linked (CGO_ENABLED=0) and embeds the
#     built SPA (go/internal/webapp's go:embed) — no native addons, no
#     node_modules. Nothing reads a font file at runtime: the shot-card
#     PNG renderer (go/internal/shots/card.go) rasterises via resvg-go on
#     wazero (pure Go/wasm, no CGo) using the Go typeface whose .TTF bytes
#     are compiled into golang.org/x/image/font/gofont — the Node app's
#     @napi-rs/canvas + system Liberation/DejaVu fonts are gone.
#   - Outbound network: the Gaggiuino/GaggiMate WebSocket and MQTT clients
#     (go/internal/machines, go/internal/mqtt) connect to whatever LAN host
#     the user configures per machine; the bean-import fetch hits a small
#     allow-listed set of HTTPS hosts; go/internal/ha calls the
#     Supervisor's own API (config.yaml's hassio_api); the
#     homeassistant_api option proxies Core API calls. All of this is plain
#     outbound TCP with normal DNS resolution — AppArmor network mediation
#     can't scope by destination host, only by address family/socket type.
#
# Strategy: a broad `file,` rule (default-deny still applies to every
# capability, mount, module load, ptrace and raw socket NOT explicitly
# granted below) rather than enumerating exact paths — kept broad because
# this repo still has no way to run real AppArmor mediation in CI and a
# too-tight profile fails silently in the field. The capability/network/
# exec lists ARE scoped tightly to what's documented above; /bin/** and
# /sbin/** ix cover Alpine's busybox shell/su-exec, which live there rather
# than under /usr/bin or /usr/local/bin.
profile gaggiuino_local_profiler flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  # App's own file I/O: /data (DB + WAL/SHM, config/options files,
  # bean-images), /tmp, /proc, and whatever the statically-linked Go binary
  # itself needs. #977 follow-up code review: no /app directory exists any
  # more (the old Node WORKDIR) — glp-server is a single file at
  # /usr/local/bin/glp-server with the frontend bundle embedded via Go's
  # embed.FS, not an on-disk code/node_modules tree.
  file,

  # Boot-time privilege drop performed by docker-entrypoint.sh: chown -R on
  # the /data bind mount, then su-exec (setuid/setgid) from root to the glp
  # user for the rest of the process's life.
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability setgid,
  capability setuid,

  # Graceful shutdown (SIGTERM/SIGINT forwarding) between the entrypoint
  # shell, su-exec, and the glp-server process it execs.
  signal,

  # Execute the entrypoint, su-exec, and glp-server itself within this same
  # profile (ix = inherit) so nothing here needs its own child profile.
  /usr/local/bin/docker-entrypoint.sh rix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /usr/local/bin/** ix,
  /bin/** ix,
  /sbin/** ix,

  # WebSocket/MQTT clients to user-configured LAN machines, HTTPS bean-
  # import fetches, Supervisor/Core API calls, and normal DNS resolution.
  # No raw/packet sockets — nothing in this codebase needs them.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,
}
